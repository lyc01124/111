const express = require('express'); const fs = require('fs'); const path = require('path'); const crypto = require('crypto'); const app = express(); const PORT = process.env.PORT || 10000; // 文件路径 const DATA_PATH = path.join(__dirname, 'data.json'); const TIME_PATH = path.join(__dirname, 'time.json'); // 初始化文件 function init() { if (!fs.existsSync(DATA_PATH)) { fs.writeFileSync(DATA_PATH, JSON.stringify([])); } if (!fs.existsSync(TIME_PATH)) { fs.writeFileSync(TIME_PATH, JSON.stringify({})); } } init(); // 缓存 2分钟 const CACHE_TTL = 120 * 1000; let cache = { list: { value: null, time: 0 } }; app.use((req, res, next) => { res.header("Access-Control-Allow-Origin", "*"); res.header("Access-Control-Allow-Methods", "GET, POST, PUT, DELETE, OPTIONS"); res.header("Access-Control-Allow-Headers", "Content-Type"); if (req.method === "OPTIONS") return res.sendStatus(200); next(); }); app.use(express.json()); app.use(express.static(path.join(__dirname, './'))); // 读取全部数据 function readAll() { const raw = fs.readFileSync(DATA_PATH, 'utf8'); return JSON.parse(raw); } // 保存全部数据 function saveAll(arr) { fs.writeFileSync(DATA_PATH, JSON.stringify(arr)); cache.list.value = arr; cache.list.time = Date.now(); } // 读取时间表 function readTimeMap() { const raw = fs.readFileSync(TIME_PATH, 'utf8'); return JSON.parse(raw); } function saveTimeMap(map) { fs.writeFileSync(TIME_PATH, JSON.stringify(map)); } // 解析时长：1F / 3S / 2H / 1T / 1Y function parsePrefix(prefix) { const m = prefix.match(/^(\d+)([F S H T Y])$/i); if (!m) return null; const num = parseInt(m[1]); const unit = m[2].toUpperCase(); const map = { S: 1000, F: 60 * 1000, H: 60 * 60 * 1000, T: 24 * 60 * 60 * 1000, Y: 365 * 24 * 60 * 60 * 1000 }; return num * map[unit]; } // 生成卡密接口 /gen app.post('/gen', (req, res) => { const { prefix } = req.body; if (!prefix) return res.json({ ok: false, msg: "请输入前缀" }); const durMs = parsePrefix(prefix); if (!durMs) return res.json({ ok: false, msg: "格式错误，示例：1F" }); const code = crypto.randomBytes(8).toString('hex')..slice(0, 10)..toUpperCase(); const all = readAll(); const id = all.length + 1; const now = Date.now(); const expireAt = now + durMs; all.push({ id, code, active: '0', activateAt: null, expireAt }); saveAll(all); const timeMap = readTimeMap(); timeMap[code] = expireAt; saveTimeMap(timeMap); return res.json({ ok: true, code }); }); // 获取列表 /list app.get('/list', (req, res) => { const now = Date.now(); if (cache.list.value && now - cache.list.time < CACHE_TTL) { return res.json({ data: cache.list.value }); } const data = readAll(); cache.list.value = data; cache.list.time = now; return res.json({ data }); }); // 格式化剩余时间 /fmt/:ms app.get('/fmt/:ms', (req, res) => { const remainMs = Number(req.params.ms); if (!remainMs || remainMs <= 0) { return res.send("已过期"); } let t = remainMs; const d = Math.floor(t / (1000*60*60*24)); t -= d * (1000*60*60*24); const h = Math.floor(t / (1000*60*60)); t -= h * (1000*60*60); const m = Math.floor(t / (1000*60)); t -= m * (1000*60); const s = Math.floor(t / 1000); const arr = []; if(d>0) arr.push(`${d}天`); if(h>0) arr.push(`${h}时`); if(m>0) arr.push(`${m}分`); if(s>0) arr.push(`${s}秒`); res.send(arr.join(' ')); }); // 删除卡密 app.post('/api/delete-code', (req, res) => { const { code } = req.body; if (!code) return res.json({ ok: false, msg: "缺少code" }); let all = readAll(); all = all.filter(item => item.code !== code); // 重新编排编号id all = all.map((it, idx) => ({ ...it, id: idx + 1 })); saveAll(all); const tm = readTimeMap(); delete tm[code]; saveTimeMap(tm); return res.json({ ok: true }); }); app.get('/', (req, res) => { res.sendFile(path.join(__dirname, 'index.html')); }); app.listen(PORT, () => { console.log(`🚀 服务启动 port:${PORT}`); });
